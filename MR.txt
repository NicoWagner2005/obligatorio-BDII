UsuarioGeneral(mail, fechaRegistro, estadoIdentidad)
    PK(mail)

Venta(idVenta, mailComprador, estado, tasaComision, montoTotal)
    PK(idVenta)
    FK(mailComprador) -> UsuarioGeneral(mail)

DetalleVenta(idVenta, nroLinea, idSector, idEvento, cantidad, precioUnitario, subTotal)
    PK(idVenta, nroLinea)
    FK(idVenta) -> Venta(idVenta)
    FK(idSector) -> Sector(idSector)
    FK(idEvento) -> Evento(idEvento)

Entrada(idEntrada, idVenta, nroLinea, mailPoseedor)
    PK(idEntrada)
    FK(idVenta, nroLinea) -> DetalleVenta(idVenta, nroLinea) 
    FK(mailPoseedor) -> UsuarioGeneral(mail)

TokenQR(idTokenQR, idEntrada, fechaExpiracion)
    PK(idTokenQR)
    FK(idEntrada) -> Entrada(idEntrada)

ValidacionAcceso(idValidacion, idTokenQR, idEntrada, idDispositivo, nroLegajo) PK(idValidacion)
    FK(idTokenQR) -> TokenQR(idTokenQR)
    FK(idEntrada) -> Entrada(idEntrada)
    FK(idDispositivo) -> DispositivoEscaneo(idDispositivo)
    FK(nroLegajo) -> Funcionario(nroLegajo)

Transferencia(idTransferencia, idEntrada, mailSolicitante, mailReceptor, estado, fechaSolicitud) 
    PK(idTransferencia)
    FK(idEntrada) -> Entrada(idEntrada)
    FK(mailSolicitante) -> UsuarioGeneral(mail) 
    FK(mailReceptor) -> UsuarioGeneral(mail)

PaisSede(nombre)
	PK(nombre)

AdministradorAsignado_aPaisSede(documentoAdmin, nombrePaisSede)
	PK(documentoAdmin)
	FK(documentoAdmin) -> Administrador(documento)
	FK(nombrePaisSede) -> PaisSede(nombre)
Estadio(idEstadio, nombre, nombrePaisSede)
	PK(idEstadio)
	FK(nombrePaisSede) -> PaisSede(nombre)
Evento(idEvento, fechaHora, documentoAdmin, idEstadio)
	PK(idEvento)
	FK(documentoAdmin) -> Administrador(documento)
	FK(idEstadio) -> Estadio(idEstadio)
Equipo(idEquipo, nombre)
	PK(idEquipo)
EquipoJuegaEvento(idEquipo, idEvento, rol)
	PK(idEquipo, idEvento)
	FK(idEquipo) -> Equipo(idEquipo)
	FK(idEvento) -> Evento(idEvento)
EventoHabilitaSector(idEvento, idEstadio, idSector)
	PK(idEvento, idEstadio, idSector)
	FK(idEvento) -> Evento(idEvento)
	FK(idEstadio) -> Sector(idEstadio)
	FK(idSector) -> Sector(idSector)
