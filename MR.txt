UsuarioGeneral(mailUsuario, fechaRegistro, estadoIdentidad)
    PK(mailUsuario)
    FK(mailUsuario) -> Usuario(mail)

Venta(idVenta, mailComprador, estado, tasaComision, montoTotal)
    PK(idVenta)
    FK(mailComprador) -> UsuarioGeneral(mail)

DetalleVenta(idVenta, nroLinea, idSector, idEvento, cantidad, precioUnitario, subTotal)
    PK(idVenta, nroLinea)
    FK(idVenta) -> Venta(idVenta)
    FK(idSector) -> Sector(idSector)
    FK(idEvento) -> Evento(idEvento)

Entrada(idEntrada, idVenta, nroLinea, mailPoseedor)
    PK(idEntrada)
    FK(idVenta, nroLinea) -> DetalleVenta(idVenta, nroLinea) 
    FK(mailPoseedor) -> UsuarioGeneral(mail)

TokenQR(idTokenQR, idEntrada, fechaExpiracion)
    PK(idTokenQR)
    FK(idEntrada) -> Entrada(idEntrada)

ValidacionAcceso(idValidacion, idTokenQR, idEntrada, mailFuncionario)
    PK(idValidacion)
    FK(idTokenQR) -> TokenQR(idTokenQR)
    FK(idEntrada) -> Entrada(idEntrada)
    FK(mailFuncionario) -> Funcionario(mailFuncionario)

Transferencia(idTransferencia, idEntrada, mailSolicitante, mailReceptor, estado, fechaSolicitud) 
    PK(idTransferencia)
    FK(idEntrada) -> Entrada(idEntrada)
    FK(mailSolicitante) -> UsuarioGeneral(mail) 
    FK(mailReceptor) -> UsuarioGeneral(mail)

Usuario(mail, nroDocumento, tipoDocumento, paisDocumento, paisDireccion, localidad, calle, nroDireccion, codigoPostal)
    PK(mail)
    AK(nroDocumento, tipoDocumento, paisDocumento)

TelefonosUsuario(mailUsuario, telefono)
    PK(mailUsuario, telefono)
    FK(mailUsuario) -> Usuario(mail)

Funcionario(mailFuncionario, nroLegajo, idDispositivo)
    PK(mailFuncionario)
    AK(nroLegajo)
    FK(mailFuncionario) -> Usuario(mail)
    FK(idDispositivo) -> DispositivoEscaneo(idDispositivo)

FuncionarioAsignadoASectorEnEvento(mailFuncionario, idEvento, idEstadio, idSector)
    PK(mailFuncionario, idEvento, idEstadio, idSector)
    FK(mailFuncionario) -> Funcionario(mailFuncionario)
    FK(idEvento) -> Evento(idEvento)
    FK(idEstadio, idSector) -> Sector(idEstadio, idSector)

Administrador(mailAdministrador, fechaAsignacion)
    PK(mailAdministrador)
    FK(mailAdministrador) -> Usuario(mail)

DispositivoEscaneo(idDispositivo)
    PK(idDispositivo)
