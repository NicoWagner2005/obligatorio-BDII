Usuario(mail, nroDocumento, tipoDocumento, paisDocumento, paisDireccion, localidad, calle, nroDireccion, codigoPostal)
    PK(mail)
    AK(nroDocumento, tipoDocumento, paisDocumento)

Funcionario(mailFuncionario, nroLegajo, idDispositivo)
    PK(mailFuncionario)
    AK(nroLegajo)
    FK(mailFuncionario) -> Usuario(mail)
    FK(idDispositivo) -> DispositivoEscaneo(idDispositivo)

Administrador(mailAdministrador, fechaAsignacion)
    PK(mailAdministrador)
    FK(mailAdministrador) -> Usuario(mail)

UsuarioGeneral(mailUsuario, fechaRegistro, estadoIdentidad)
PK(mailUsuario)
FK(mailUsuario) -> Usuario(mail)
-- fechaRegistro NOT NULL
-- estadoIdentidad NOT NULL

Venta(idVenta, mailComprador, estado, tasaComision, montoTotal, fecha)
PK(idVenta)
FK(mailComprador) -> UsuarioGeneral(mailUsuario)
-- mailComprador NOT NULL
-- estado NOT NULL
-- fecha NOT NULL

DetalleVenta(idVenta, nroLinea, idEvento, idEstadio, idSector, cantidad, subTotal)
PK(idVenta, nroLinea)
FK(idVenta) -> Venta(idVenta) # La compra de la entrada es para un sector de un evento en un estadio
FK(idEvento, idEstadio, idSector) -> EventoHabilitaSector(idEvento, idEstadio, idSector)
-- idEvento NOT NULL
-- idEstadio NOT NULL
-- idSector NOT NULL
-- cantidad NOT NULL
RNE: cantidad ≤ 5

Transferencia(idTransferencia, idEntrada, mailSolicitante, mailReceptor, estado, fechaSolicitud)
PK(idTransferencia)
FK(idEntrada) -> Entrada(idEntrada)
FK(mailSolicitante) -> UsuarioGeneral(mailUsuario)
FK(mailReceptor) -> UsuarioGeneral(mailUsuario)
-- idEntrada NOT NULL
-- mailSolicitante NOT NULL
-- mailReceptor NOT NULL
-- estado NOT NULL
-- fechaSolicitud NOT NULL
RNE: Una entrada puede transferirse como máximo 3 veces
RNE: mailSolicitante ≠ mailReceptor
